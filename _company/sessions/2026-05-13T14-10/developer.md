import os

class LeadMagnetPDF:
    def __init__(self, theme_name):
        from fpdf import FPDF
        self.pdf = FPDF()
        self.theme_name = theme_name
        self.themes = {
            "ARCHIVE": {"bg": (245, 245, 220), "text": (40, 40, 40), "accent": (184, 134, 11), "header": (101, 67, 33)},
            "MODERN": {"bg": (255, 255, 255), "text": (20, 20, 20), "accent": (220, 20, 60), "header": (0, 0, 0)},
            "COMMUNITY": {"bg": (240, 248, 255), "text": (30, 30, 80), "accent": (70, 130, 180), "header": (0, 51, 102)}
        }
        self.colors = self.themes.get(theme_name, self.themes["MODERN"])
        self.pdf.set_auto_page_break(auto=True, margin=15)

    def add_page(self):
        self.pdf.add_page()
        self.pdf.set_fill_color(*self.colors["bg"])
        self.pdf.rect(0, 0, 210, 297, 'F')
        
        # Header
        self.pdf.set_fill_color(*self.colors["header"])
        self.pdf.rect(0, 0, 210, 25, 'F')
        self.pdf.set_font("Arial", 'B', 14)
        self.pdf.set_text_color(*self.colors["bg"])
        self.pdf.cell(0, 15, "DATASSY LEAD MAGNET", 0, 1, 'C')
        self.pdf.ln(10)

    def add_content(self, markdown_text):
        self.pdf.set_font("Arial", '', 12)
        self.pdf.set_text_color(*self.colors["text"])
        
        lines = markdown_text.split('\n')
        for line in lines:
            if line.startswith('# '):
                self.pdf.ln(5)
                self.pdf.set_font("Arial", 'B', 18)
                self.pdf.set_text_color(*self.colors["accent"])
                self.pdf.cell(0, 10, line[2:], 0, 1, 'L')
                self.pdf.set_font("Arial", '', 12)
                self.pdf.set_text_color(*self.colors["text"])
            elif line.startswith('## '):
                self.pdf.ln(5)
                self.pdf.set_font("Arial", 'B', 14)
                self.pdf.set_text_color(*self.colors["accent"])
                self.pdf_cell_width = 0
                self.pdf.cell(0, 10, line[3:], 0, 1, 'L')
                self.pdf.set_font("Arial", '', 12)
                self.pdf.set_text_color(*self.colors["text"])
            elif line.strip() == "":
                self.pdf.ln(2)
            else:
                self.pdf.multi_cell(0, 8, line.replace('**', ''), 0, 'L')

    def add_footer(self):
        self.pdf.set_y(-15)
        self.pdf.set_font("Arial", 'I', 8)
        self.pdf.set_text_color(128, 128, 128)
        self.pdf.cell(0, 10, f"Generated by Datassy AI - Theme: {self.theme_name}", 0, 0, 'C')

    def save(self, output_path):
        self.pdf.output(output_path)
