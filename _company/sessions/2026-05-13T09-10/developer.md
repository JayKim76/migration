import os

class LeadMagnetFormatter:
    """Researcher의 리서치 데이터를 Lead Magnet 규격으로 변환하는 도구"""
    
    def __init__(self, output_dir="projects/automation_dashboard/outputs"):
        self.output_dir = output_dir
        if not os.path.exists(self.output_dir):
            os.makedirs(self.output_dir)

    def format_to_lead_magnet(self, input_path, title):
        """Markdown 파일을 읽어 Lead Magnet용 헤더를 추가하고 저장"""
        if not os.path.exists(input_path):
            return f"Error: {input_path} 파일을 찾을 수 없습니다."

        with open(input_path, 't', encoding='utf-8') as f:
            content = f.read()

        header = f"# 🚀 [LEAD MAGNET] {title}\n\n"
        header += f"**Date:** Generated by Datasys Automation\n"
        header += f"**Status:** Verified by AI Agent\n"
        header += "---\n\n"
        
        new_content = header + content
        output_filename = f"{title.replace(' ', '_').lower()}_magnet.md"
        output_path = os.path.join(self.output_dir, output_filename)

        with open(output_path, 'w', encoding='utf-8') as f:
            f.write(new_content)
        
        return f"Success: Lead Magnet 생성 완료 -> {output_path}"

if __name__ == "__main__":
    # 테스트 실행 (파일이 있을 경우)
    formatter = LeadMagnetFormatter()
    print(formatter.format_to_lead_magnet("test_research.md", "AI Automation Checklist"))
